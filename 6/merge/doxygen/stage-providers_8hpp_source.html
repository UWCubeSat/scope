35inline CameraParameters ProvideCameraParameters(const RecalibrationOptions &options) {
36 // A zero focal length projects every star onto the principal point, so
37 // stage 2 would quietly find nothing.
38 if (!options.starImages.empty() && !(options.focalLengthX > DECIMAL(0.0) && options.focalLengthY > DECIMAL(0.0))) {
39 throw std::runtime_error("--focal-length-x and --focal-length-y must be positive");
40 }
41
42 CameraParameters camera;
43 camera.focalLengthX = options.focalLengthX;
44 camera.alpha = options.alpha;
45 camera.focalLengthY = options.focalLengthY;
46 camera.principalX = options.principalX;
47 camera.principalY = options.principalY;
48 camera.k1 = options.k1;
49 camera.k2 = options.k2;
50 camera.k3 = options.k3;
51 camera.p1 = options.p1;
52 camera.p2 = options.p2;
53 return camera;
54}

120inline FitSettings ProvideFitSettings(const RecalibrationOptions &options) {
121 FitSettings settings;
122
123 settings.estimate.fill(true);
124 std::string names = options.fixedParameters;
125 std::replace(names.begin(), names.end(), ',', ' ');
126 std::istringstream list(names);
127 std::string name;
128 while (list >> name) {
129 if (name == "none") {
130 continue;
131 }
132 if (name == "all") {
133 settings.estimate.fill(false);
134 continue;
135 }
136 const auto match = std::find(kCameraParameterNames.begin(), kCameraParameterNames.end(), name);
137 if (match == kCameraParameterNames.end()) {
138 throw std::runtime_error("--fixed-parameters: unknown camera parameter '" + name + "'");
139 }
140 settings.estimate[static_cast<std::size_t>(match - kCameraParameterNames.begin())] = false;
141 }
142
143 if (options.attitudeCorrection == "per-image") {
144 settings.attitudeModel = AttitudeCorrectionModel::kPerImage;
145 } else if (options.attitudeCorrection == "shared") {
146 settings.attitudeModel = AttitudeCorrectionModel::kShared;
147 } else {
148 throw std::runtime_error("--attitude-correction must be 'per-image' or 'shared', not '" +
149 options.attitudeCorrection + "'");
150 }
151
152 if (!(options.outlierThreshold >= DECIMAL(0.0))) {
153 throw std::runtime_error("--outlier-threshold must not be negative");
154 }
155 settings.outlierThreshold = options.outlierThreshold;
156 if (options.maxIterations < 1) {
157 throw std::runtime_error("--max-iterations must be at least 1");
158 }
159 settings.maxIterations = options.maxIterations;
160
161 return settings;
162}
